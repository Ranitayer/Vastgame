cmake_minimum_required(VERSION 3.16)
project(vastgame_crashpad LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CRASHPAD_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(CRASHPAD_ENABLE_INSTALL_DEV OFF CACHE BOOL "" FORCE)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "This module supports native Linux Moonlight only")
endif()
if(NOT CRASHPAD_SOURCE)
    message(FATAL_ERROR "Set CRASHPAD_SOURCE to the pinned getsentry/crashpad checkout")
endif()
add_subdirectory(${CRASHPAD_SOURCE} crashpad)
add_library(vastgame_crashpad SHARED crashpad.cpp)
target_compile_features(vastgame_crashpad PRIVATE cxx_std_17)
target_link_libraries(vastgame_crashpad PRIVATE crashpad_client)
target_link_options(vastgame_crashpad PRIVATE "-Wl,--exclude-libs,ALL")
set_target_properties(vastgame_crashpad PROPERTIES PREFIX "")
